Testing Electronic Components Guide

Comprehensive guide on testing Resistors, Capacitors, Diodes, LEDs, and Voltage Sources using a Digital Multimeter (DMM) and Analog Volt-Ohm-Milliammeter (VOM).

Mandatory Safety Rules:
1. Power Down the Circuit completely before measuring resistance, capacitance, or diodes.
2. Discharge Capacitors prior to touching or measuring.
3. Isolate Components (desolder at least one lead for accurate out-of-circuit testing).
4. Verify Voltmeter settings (AC vs. DC) before measuring live voltage circuits!

1. Resistor Diagnostics

Resistors limit current (R = V/I). Check color codes or SMD labels for nominal value and tolerance.

  • Set DMM to Resistance (Ω). Touch probes to leads (non-polar).
  • Good: Reading within tolerance (e.g., 1 kΩ ± 5% = 950 Ω – 1050 Ω).
  • Defective: Displays OL (Open) or drifted significantly higher due to thermal degradation.

2. Capacitor Diagnostics

Capacitors store electrical energy electrostatically and filter power line noise. Discharge fully before testing!

  • Capacitance Mode: Connect red probe to positive anode (longer lead) and black to cathode. Confirm value within ±20% of rated value in Farads (F / µF).
  • Resistance Mode Charging Check: Set to high resistance (100 kΩ). Probe should show low resistance, then steadily rise toward infinity (OL) as the meter battery charges the dielectric.

3. Diode & LED Diagnostics

One-way semiconductor gates. Set DMM to Diode Mode |.

  • Forward Bias (Red to Anode, Black to Cathode): Silicon diodes drop 0.5 V – 0.7 V; Schottky diodes drop 0.15 V – 0.4 V; LEDs drop 1.8 V – 3.5 V and glow faintly.
  • Reverse Bias (Red to Cathode, Black to Anode): Must read OL (Over Limit). Reading a low voltage drop in reverse bias indicates a shorted or leaky semiconductor junction.

Backend Source Code (`Code.gs`)

/**
 * ==============================================================================
 * GOOGLE APPS SCRIPT BACKEND SERVICE
 * Handles Assessment Submissions and Voluntary Donation Records with
 * Idempotency Locking to prevent duplicate records.
 * ==============================================================================
 */

// Global Worksheet Tab Names
const TAB_ASSESSMENT = "AssessmentSubmissions";
const TAB_DONATION = "DonationRecords";

/**
 * Handles incoming POST requests from the Educational Web Application.
 */
function doPost(e) {
  const lock = LockService.getScriptLock();
  // Acquire lock with 10-second timeout to handle concurrent requests safely
  try {
    lock.waitLock(10000);
  } catch (err) {
    return createJsonResponse({ status: "error", message: "Server busy. Please try submitting again." });
  }

  try {
    if (!e || !e.postData || !e.postData.contents) {
      return createJsonResponse({ status: "error", message: "Empty submission payload." });
    }

    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // Route request based on payload type
    if (data.type === "ASSESSMENT") {
      return processAssessmentSubmission(ss, data);
    } else if (data.type === "DONATION") {
      return processDonationRecord(ss, data);
    } else {
      return createJsonResponse({ status: "error", message: "Unknown payload type." });
    }

  } catch (error) {
    return createJsonResponse({ status: "error", message: error.toString() });
  } finally {
    lock.releaseLock();
  }
}

/**
 * Processes and appends assessment records with duplicate checking.
 */
function processAssessmentSubmission(ss, data) {
  let sheet = ss.getSheetByName(TAB_ASSESSMENT);
  if (!sheet) {
    sheet = initializeAssessmentSheet(ss);
  }

  // Idempotency Check: Prevent duplicate Submission IDs
  const submissionId = data.submissionId;
  const existingIds = sheet.getRange(2, 2, Math.max(sheet.getLastRow() - 1, 1), 1).getValues().flat();
  
  if (existingIds.indexOf(submissionId) !== -1) {
    return createJsonResponse({ 
      status: "success", 
      message: "Submission already recorded previously (Duplicate request ignored)." 
    });
  }

  // Append new assessment record
  const timestamp = new Date();
  sheet.appendRow([
    timestamp,
    submissionId,
    data.fullName,
    data.gradeLevel,
    data.section,
    "Testing Electronic Components Knowledge Test",
    data.totalItems,
    data.correctAnswers,
    data.incorrectAnswers,
    data.score,
    data.percentage,
    "RECORDED"
  ]);

  return createJsonResponse({ status: "success", message: "Assessment recorded successfully." });
}

/**
 * Processes and appends donation records.
 */
function processDonationRecord(ss, data) {
  let sheet = ss.getSheetByName(TAB_DONATION);
  if (!sheet) {
    sheet = initializeDonationSheet(ss);
  }

  const timestamp = new Date();
  sheet.appendRow([
    timestamp,
    data.donationId,
    data.donorName,
    data.amount,
    data.currency || "PHP",
    "SUBMITTED"
  ]);

  return createJsonResponse({ status: "success", message: "Donation record logged." });
}

/**
 * Initializes Assessment Worksheet structure automatically.
 */
function initializeAssessmentSheet(ss) {
  const sheet = ss.insertSheet(TAB_ASSESSMENT);
  const headers = [
    "Timestamp", "Submission ID", "Full Name", "Grade Level", "Section",
    "Assessment Name", "Total Items", "Correct Answers", "Incorrect Answers",
    "Score", "Percentage", "Status"
  ];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight("bold").setBackground("#f3f4f6");
  sheet.setFrozenRows(1);
  return sheet;
}

/**
 * Initializes Donation Worksheet structure automatically.
 */
function initializeDonationSheet(ss) {
  const sheet = ss.insertSheet(TAB_DONATION);
  const headers = ["Timestamp", "Donation ID", "Donor Name", "Amount (PHP)", "Currency", "Status"];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight("bold").setBackground("#f3f4f6");
  sheet.setFrozenRows(1);
  return sheet;
}

/**
 * Helper to construct JSON response output.
 */
function createJsonResponse(responseObject) {
  return ContentService.createTextOutput(JSON.stringify(responseObject))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * Health Check Endpoint (doGet)
 */
function doGet(e) {
  return ContentService.createTextOutput("Testing Electronic Components Web App Backend is Active.");
}
